    <script>
        let cart = [];
        let selectedSize = 'S';
        
        // Sample products list matching the Lab UI
        const products = [
            { name: "Club Swim Short in Volt", price: 60, img: "https://images.unsplash.com/photo-1591195853828-11db59a44f6b?w=500" },
            { name: "Heavy Loopback Short", price: 65, img: "https://images.unsplash.com/photo-1562157873-818bc0726f68?w=500" },
            { name: "Tech Fleece Set", price: 240, img: "https://images.unsplash.com/photo-1552374196-1ab2a1c593e8?w=500" },
            { name: "Piped Woven Tracksuit", price: 210, img: "https://images.unsplash.com/photo-1515886657613-9f3515b0c78f?w=500" }
        ];
        let currentIndex = 0;

        function updateProductView() {
            let p = products[currentIndex];
            document.getElementById('mainProductImg').src = p.img;
            document.getElementById('itemName').innerText = p.name;
            document.getElementById('itemPrice').innerText = 'EGP ' + p.price;
            document.getElementById('itemCounter').innerText = `0${currentIndex + 1} / 11`;
        }

        function nextProduct() {
            currentIndex = (currentIndex + 1) % products.length;
            updateProductView();
        }

        function prevProduct() {
            currentIndex = (currentIndex - 1 + products.length) % products.length;
            updateProductView();
        }

        function selectSize(btn) {
            document.querySelectorAll('.size-btn').forEach(b => b.classList.remove('selected'));
            btn.classList.add('selected');
            selectedSize = btn.innerText;
        }

        function changeColor(colorHex) {
            document.querySelectorAll('.color-dot').forEach(d => d.classList.remove('active'));
            event.target.classList.add('active');
        }

        function addToCart() {
            let p = products[currentIndex];
            cart.push({ ...p, size: selectedSize });
            document.getElementById('cartCount').innerText = cart.length;
            document.getElementById('cartDrawerCount').innerText = cart.length;
            updateCartUI();
            alert("Added to bag successfully!");
        }

        function toggleCart() {
            document.getElementById('cartDrawer').classList.toggle('open');
        }

        function updateCartUI() {
            let list = document.getElementById('cartItemsList');
            if (cart.length === 0) {
                list.innerHTML = "Your bag is empty.";
                return;
            }
            let html = "";
            let total = 0;
            cart.forEach((item, idx) => {
                total += item.price;
                html += `<div style="display:flex; justify-content:space-between; margin-bottom:8px;">
                    <span>${item.name} (${item.size})</span>
                    <span>EGP ${item.price}</span>
                </div>`;
            });
            html += `<hr style="border-color:#444; margin:10px 0;"><strong>Total: EGP ${total}</strong>`;
            list.innerHTML = html;
        }

        function sendWhatsAppOrder() {
            let name = document.getElementById('custName').value;
            let phone = document.getElementById('custPhone').value;
            let address = document.getElementById('custAddress').value;

            if(!name || !phone || !address) {
                alert("Please fill in your name, WhatsApp number, and address.");
                return;
            }

            let orderText = `New Order from XSTREET:\nName: ${name}\nPhone: ${phone}\nAddress: ${address}\n\nItems:\n`;
            let total = 0;
            cart.forEach(item => {
                orderText += `- ${item.name} | Size: ${item.size} | EGP ${item.price}\n`;
                total += item.price;
            });
            orderText += `\nTotal Amount: EGP ${total}`;

            let encodedText = encodeURIComponent(orderText);
            let whatsappUrl = `https://wa.me/201551102291?text=${encodedText}`;
            window.open(whatsappUrl, '_blank');
        }

        function filterCategory(cat) {
            document.querySelectorAll('.cat-chip').forEach(c => c.classList.remove('active'));
            event.target.classList.add('active');
        }
    </script>
